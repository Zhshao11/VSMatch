## 1. CMR@1 / CMR@2 / CMR@5 的计算

### 1.1 官方定义（`OFFICIAL/test.py`）

```python
def Registration_index_calculation(epe_list, T):
    epe_values_less_than_T = [value for value in epe_list if value < T]
    MAE = np.mean(epe_values_less_than_T) if epe_values_less_than_T else 0
    RMSE = np.var(epe_values_less_than_T) if epe_values_less_than_T else 0
    CMR = 100 * (len(epe_values_less_than_T) / len(epe_list))
    return MAE, RMSE, CMR

# validate() 内部
MAE_list = []          # 长度 = 424，元素 EPE(k) = 每对图像的平均误差
for val_id in range(len(val_dataset)):
    ...
    epe = torch.sum((flow_pre - flow_gt)**2, dim=0).sqrt()   # [2,512,512]
    epe_ = np.mean(epe.view(-1).numpy())
    MAE_list.append(epe_)

MAE_1, RMSE_1, CMR_1 = Registration_index_calculation(MAE_list, 1)
MAE_2, RMSE_2, CMR_2 = Registration_index_calculation(MAE_list, 2)
MAE_5, RMSE_5, CMR_5 = Registration_index_calculation(MAE_list, 5)
```

即 **配对级**：`CMR@τ = 100 × |{k : EPE(k) < τ}| / N_pairs`（`<` 严格小于，不是 `≤`）。
同一函数返回的 `MAE` 就是 `AEPE@τ`，`RMSE` 是 `np.var(inliers)`（见第 2 部分）。

## 2. AEPE / RMSE，以及 AEPE@τ 与全局 AEPE 的区别

### 2.1 官方口径（`OFFICIAL/test.py` + `LOCAL/eval_os_official_protocol.py`）

```python
# 每对先算 EPE(k)
e_comp = torch.sum(diff ** 2, dim=0).sqrt().reshape(-1)   # [2*512*512] 逐分量 |dx|,|dy|
EPE_k  = float(e_comp.mean())                            # -> MAE_list[k]

p = np.asarray(MAE_list, dtype=np.float64)                 # shape (424,)

AEPE_all = p.mean()          # ★ 全局 AEPE：全部 424 对，一个都不剔除
RMSE_official = p.var()      # ★ 官方 test.py 打印的 "RMSE" 实为方差（无 sqrt）
RMSE_paper_eq15 = p.std()    # 论文公式 (15) 写的是 sqrt(var)，与代码不一致

# AEPE@τ：只在通过阈值的配对上求均值
for t in (1, 2, 5):
    inl = p[p < t]                       # 子集，不是全体
    CMR_at_t     = 100.0 * inl.size / p.size
    AEPE_at_t    = inl.mean()             # ★ 恒 < t；样本集随 τ 变化
    RMSE_at_t    = inl.var()              # 官方 T 档 "RMSE" 同样是方差
```

**核心区别（容易误读的地方）**：

| | 样本集 | 取值 | 受失败配对影响 |
|---|---|---|---|
| 全局 `AEPE` | 全部 424 对 | 可为任意正数 | 直接被灾难配对抬高 |
| `AEPE@τ` | 仅 `EPE(k)<τ` 的配对 | 恒 `< τ` | **失败配对被筛掉**，只反映成功子集精度 |
| `CMR@τ` | 同上，返回占比 | 0–100% | 与 `AEPE@τ` 必须联读 |

所以 `AEPE@τ` 低不等于模型整体好：CMR 低时，它只是少数"简单配对"的均值。论文原文也写明
`AEPE@τ is threshold-dependent … should not be interpreted in isolation`。
本次实测即为例：M4 全局 AEPE 1.4738（几乎持平 GDROS 1.4618），但 `AEPE@5` 只有 0.8596
（GDROS 1.4431），因为 M4 的 CMR@5 只有 94.10%，5.9% 的灾难配对被排除在 `AEPE@5` 之外，
却全部计入全局 AEPE 与方差 → 官方 RMSE 8.3350 vs 0.7463。

### 3.2 GRU 在 1/8 分辨率出流 → 上采样回原图（`TMP/e073a_v1_v2_fusion_longtrain.py`）

```python
def initialize_flow(self, img):
    n, _, h, w = img.shape
    return coords_grid(n, h // 8, w // 8, ...), coords_grid(n, h // 8, w // 8, ...)  # 64x64

def upsample_flow(self, flow, mask):
    n, _, h, w = flow.shape                      # [N,2,64,64]
    if mask is None:
        return F.interpolate(8 * flow, scale_factor=8, mode="bilinear", align_corners=True)
    mask = torch.softmax(mask.view(n, 1, 9, 8, 8, h, w), dim=2)
    up = F.unfold(8 * flow, [3, 3], padding=1).view(n, 2, 9, 1, 1, h, w)
    up = torch.sum(mask * up, dim=2).permute(0, 1, 4, 2, 5, 3).reshape(n, 2, 8 * h, 8 * w)
    return up                                     # [N,2,512,512]

def recurrent_from_corr(self, img1n, corr_fn):
    ...
    for _ in range(self.iters):                  # iters=32
        c1 = c1.detach(); corr = corr_fn(c1)
        net, mask, delta = self.update(net, inp, corr, c1 - c0)
        c1 = c1 + delta
        outs.append(self.upsample_flow(c1 - c0, mask))
    return outs                                   # 32 个全分辨率流
```

`8 * flow` 把低分辨率坐标系的位移换算成原图像素，`scale_factor=8` 把 64×64 插值成 512×512。
输入 512 本身是 8 的倍数，**无需 InputPadder / crop / resize**。
（官方 `OFFICIAL/core/utils/utils.py` 的 `upflow8` 同义：`8 * F.interpolate(flow, size=(8h,8w), ...)`。）

class PairFusion(nn.Module):
    def forward(self, a, b, iters=12, lsr_fn=None):
        pb = self.b(a, b, iters=iters)            # E074-B 序列，每个 [1,2,512,512]
        pc = self.c(a, b, iters=iters)            # E074-C 序列
        outs = []
        for xb, xc in zip(pb, pc):
            feat = self.gate_features(xb, xc, lsr_fn)     # 16 通道证据（含各自 LSR 残差）
            g = self.gate(feat)                            # [1,1,512,512]，4x4 区域平滑
            outs.append((1 - g) * xb + g * xc)             # 仍是 512x512
        return outs
```

# 官方 LSR()
def LSR(bs, h, w, flow_pre1, Npoint):
    grid__ = get_grid(1, h, w).permute(0, 2, 3, 1)          # [1,512,512,3] 原图坐标网格
    H_flow_f = grid__[:, :, :, :2] * 0
    grid = grid__.reshape(bs, -1, 3)
    input_flow_cat = flow_pre1.permute(0, 2, 3, 1).reshape(bs, -1, 2)[:, 10000:(10000 + Npoint)]
    grid_cat      = grid[:, 10000:(10000 + Npoint), :]       # 取 20 万个点
    trans_mat = LSR_(grid_cat, input_flow_cat)               # 无权最小二乘，6 参数仿射
    for k in range(bs):
        pre_H = trans_mat[k, :, :]
        pre_H[0, 0] += 1; pre_H[1, 1] += 1                    # 残差 -> 绝对位移
        pre_H1 = torch.cat((pre_H, torch.tensor([[0, 0, 1]]).cuda()), dim=0)
        pre_H1 = pre_H1.unsqueeze(0).repeat(h * w, 1, 1, 1).unsqueeze(0)
        grid_  = grid__[0, :, :, :].reshape(-1, 3).unsqueeze(0).unsqueeze(3).float()
        grid_warp = torch.matmul(pre_H1, grid_)
        grid_warp = grid_warp.squeeze().reshape(h, w, 3).unsqueeze(0)
        flow_f = grid_warp[:, :, :, :2] / grid_warp[:, :, :, 2:] - grid__[:, :, :, :2]
        H_flow_f = torch.cat((H_flow_f, flow_f), 0)
    H_flow_f = H_flow_f[1:, ...]
    return H_flow_f.permute(0, 3, 1, 2).cpu()                # [1,2,512,512]，CPU
```

### 3.5 最终与 GT 计算误差（`LOCAL/eval_os_official_protocol.py`）

```python
@torch.inference_mode()
def run_one(...):
    for i in range(n):
        _, a_img, b_img, gt, valid = ds[i]        # gt: [2,512,512] CPU
        a = a_img[None].to(device); b = b_img[None].to(device)
        preds = model(a, b, iters=32, lsr_fn=e074.lsr)   # PairFusion, list of 512x512 flows
        post = e074.lsr(preds[-1])                       # [1,2,512,512] CPU
        diff = post - gt[None]                           # [1,2,512,512]（GT 广播）

class Collector:
    def add_pair(self, diff):
        # A) 官方 test.py 口径：dim=0 求和 -> 逐分量 |dx|,|dy|
        e_comp = torch.sum(diff ** 2, dim=0).sqrt().reshape(-1).double().numpy()
        # B) 项目内部口径：真向量 EPE
        e_vec = torch.linalg.vector_norm(diff.float(), dim=1).reshape(-1).double().numpy()
        self.pairs.append(float(e_comp.mean()))            # EPE(k) -> MAE_list
        self.vec_pairs.append(float(e_vec.mean()))
```

`post` 与 `gt` 同为 512×512，**没有任何额外的缩放、裁剪或对齐**；
`valid` 实测全 1（官方 `test.py` 直接忽略它），所以 masked 与 unmasked 统计等价。
注意 `diff` 的形状是 `[1,2,512,512]` 而 `gt` 是 `[2,512,512]` —— 官方 `dim=0` 求和之所以
得到分量误差而非端点误差，正是这个 batch 维广播造成的（详见 `GDROS复现口径核查_20261113.md`）。
